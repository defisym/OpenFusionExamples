cmake_minimum_required(VERSION 3.22)
project(FiniJNI LANGUAGES CXX)
# Publisher-only native build. Never replaces Fusion's Gradle templates.
if(ANDROID AND NOT OFE_ENCRYPTION_LIBRARY)
    if(NOT FINI_OPENSSL_ROOT)
        message(FATAL_ERROR "FINI_OPENSSL_ROOT must name the target ABI's OpenSSL prefix")
    endif()
    set(OPENSSL_INCLUDE_DIR "${FINI_OPENSSL_ROOT}/include" CACHE PATH "Target headers" FORCE)
    set(OPENSSL_CRYPTO_LIBRARY "${FINI_OPENSSL_ROOT}/lib/libcrypto.a" CACHE FILEPATH "Target static crypto" FORCE)
    if(NOT EXISTS "${OPENSSL_INCLUDE_DIR}/openssl/evp.h" OR NOT EXISTS "${OPENSSL_CRYPTO_LIBRARY}")
        message(FATAL_ERROR "Missing target OpenSSL headers or libcrypto.a")
    endif()
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../Fini/Mobile" core)
add_library(fini SHARED ../../app/src/main/cpp/FiniJNI.cpp)
target_link_libraries(fini PRIVATE fini_core)
if(ANDROID)
    target_link_libraries(fini PRIVATE dl)
    target_link_options(fini PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_path(FINI_JNI jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(FINI_JNI_PLATFORM jni_md.h HINTS "$ENV{JAVA_HOME}/include/linux" "$ENV{JAVA_HOME}/include/darwin" "$ENV{JAVA_HOME}/include/win32" REQUIRED)
    target_include_directories(fini PRIVATE ${FINI_JNI} ${FINI_JNI_PLATFORM})
endif()
