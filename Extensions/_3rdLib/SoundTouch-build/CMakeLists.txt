cmake_minimum_required(VERSION 3.21)
project(OFESoundTouchSDK VERSION 2.3.3 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This SoundTouch SDK is for Windows; use MSVC or a Windows cross toolchain.")
endif()
get_filename_component(source "${CMAKE_CURRENT_LIST_DIR}/../SoundTouch" ABSOLUTE)
if(NOT EXISTS "${source}/CMakeLists.txt")
    message(FATAL_ERROR "Initialize the SoundTouch submodule before configuring this SDK.")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(sdk_arch x64)
else()
    set(sdk_arch x86)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build a static core inside the DLL wrapper" FORCE)
set(SOUNDTOUCH_DLL ON CACHE BOOL "Build the C DLL interface used by OFE" FORCE)
set(INTEGER_SAMPLES OFF CACHE BOOL "The DLL wrapper requires float internals" FORCE)
set(SOUNDSTRETCH OFF CACHE BOOL "No command-line utility is required" FORCE)
set(OPENMP OFF CACHE BOOL "No additional OpenMP runtime DLL" FORCE)
# Permit the pinned upstream's CMake 3.1 policy baseline with newer CMake versions.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
add_subdirectory("${source}" "${CMAKE_BINARY_DIR}/upstream" EXCLUDE_FROM_ALL)
if(MSVC)
    # Upstream 2.3.3 puts these options in COMPILE_DEFINITIONS. Keep the source
    # unmodified and move them to compile options in this integration layer.
    get_target_property(definitions SoundTouch COMPILE_DEFINITIONS)
    list(REMOVE_ITEM definitions /O2 /fp:fast)
    set_property(TARGET SoundTouch PROPERTY COMPILE_DEFINITIONS "${definitions}")
    target_compile_options(SoundTouch PRIVATE /O2 /fp:fast)
endif()
target_compile_definitions(SoundTouchDLL PRIVATE SOUNDTOUCH_FLOAT_SAMPLES)
set_target_properties(SoundTouchDLL PROPERTIES PREFIX "" OUTPUT_NAME SoundTouchDLL)
if(MINGW)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        # Match the SSE2 baseline of the MSVC Win32 build for upstream intrinsics.
        target_compile_options(SoundTouch PRIVATE -msse2)
    endif()
    target_link_options(SoundTouchDLL PRIVATE -static -static-libgcc -static-libstdc++)
endif()
# VS builds a named target through <target>.vcxproj in the requested build
# directory. The upstream DLL project lives in upstream/, so expose a root target.
add_custom_target(soundtouch_sdk DEPENDS SoundTouchDLL)
install(TARGETS SoundTouchDLL
    RUNTIME DESTINATION "lib/${sdk_arch}"
    ARCHIVE DESTINATION "lib/${sdk_arch}")
install(FILES "${source}/source/SoundTouchDLL/SoundTouchDLL.h" DESTINATION include)
install(FILES "${source}/COPYING.TXT" DESTINATION share/licenses/SoundTouch)
