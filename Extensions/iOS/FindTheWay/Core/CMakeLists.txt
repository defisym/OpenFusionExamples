cmake_minimum_required(VERSION 3.22)
project(FindTheWayCore LANGUAGES C CXX)
set(FTW_CORE_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../_DeLib" CACHE PATH "Current shared core headers")
if(NOT EXISTS "${FTW_CORE_INCLUDE}/FindTheWayFusion.h")
    message(FATAL_ERROR "Shared core is outdated: select master 29edeed0 or newer, or set FTW_CORE_INCLUDE")
endif()
find_package(Iconv REQUIRED)
add_library(FindTheWayCore STATIC FindTheWayCore.cpp)
target_compile_features(FindTheWayCore PRIVATE cxx_std_20)
target_include_directories(FindTheWayCore PUBLIC include PRIVATE "${FTW_CORE_INCLUDE}" "${FTW_CORE_INCLUDE}/../_3rdLib")
target_link_libraries(FindTheWayCore PRIVATE Iconv::Iconv)
target_compile_definitions(FindTheWayCore PRIVATE DELIB_ACP_ENCODING="UTF-8")
set_target_properties(FindTheWayCore PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(abi_test tests/abi_test.c)
    target_link_libraries(abi_test PRIVATE FindTheWayCore)
    set_target_properties(abi_test PROPERTIES LINKER_LANGUAGE CXX)
    add_test(NAME abi_test COMMAND abi_test)
endif()
