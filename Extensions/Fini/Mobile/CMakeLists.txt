cmake_minimum_required(VERSION 3.22)
project(FiniMobile LANGUAGES CXX)
if(OFE_ENCRYPTION_LIBRARY)
    add_library(ofe_encryption SHARED IMPORTED)
    set_target_properties(ofe_encryption PROPERTIES IMPORTED_LOCATION "${OFE_ENCRYPTION_LIBRARY}")
elseif(NOT TARGET ofe_encryption)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../_DeLib/Encryption/Mobile" common-encryption)
endif()
find_package(ZLIB REQUIRED)
add_library(fini_core INTERFACE)
target_compile_features(fini_core INTERFACE cxx_std_20)
target_include_directories(fini_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../_DeLib")
target_compile_definitions(fini_core INTERFACE _FILE_OFFSET_BITS=64)
target_link_libraries(fini_core INTERFACE ofe_encryption ZLIB::ZLIB)
option(FINI_BUILD_TESTS "Build host core tests" OFF)
if(FINI_BUILD_TESTS)
    enable_testing()
    add_executable(fini_tests ../../Android_Gradle/tests/fini/core.cpp)
    target_link_libraries(fini_tests PRIVATE fini_core)
    add_test(NAME fini_core COMMAND fini_tests)
endif()
