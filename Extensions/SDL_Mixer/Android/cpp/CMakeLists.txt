cmake_minimum_required(VERSION 3.22)
project(OFESDLMixerAndroid LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
foreach(dependency SDL MIXERX SOUNDTOUCH)
    if(NOT EXISTS "${OFE_${dependency}_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "Set OFE_${dependency}_SOURCE to the pinned upstream source directory")
    endif()
endforeach()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
# Keep the stock Android JNI dependencies compiled; only AUDIO|TIMER are initialized.
if(NOT ANDROID)
    set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
    set(SDL_RENDER OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${OFE_SDL_SOURCE}" sdl)
# Mixer X expects find_package(SDL2); both projects are built together without installation.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/sdl-find")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdl-find/SDL2Config.cmake" "set(SDL2_FOUND TRUE)\nset(SDL2_LIBRARIES SDL2::SDL2)\n")
set(SDL2_DIR "${CMAKE_CURRENT_BINARY_DIR}/sdl-find" CACHE PATH "" FORCE)
set(USE_SYSTEM_SDL2 ON CACHE BOOL "" FORCE)
set(SDL_MIXER_X_SHARED ON CACHE BOOL "" FORCE)
set(SDL_MIXER_X_STATIC OFF CACHE BOOL "" FORCE)
set(MIXERX_DISABLE_SO_VERSION ON CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# Match Windows' LGPL/non-GPL codec selection. Built-in-only bootstrap builds
# SDL before the static AudioCodecs stage is available.
set(OFE_AUDIOCODECS_ROOT "" CACHE PATH "Prepared static AudioCodecs/SDL SDK")
foreach(feature OPUS WAVPACK MODPLUG XMP GME MIDI_EDMIDI MIDI_FLUIDLITE)
    if(OFE_AUDIOCODECS_ROOT)
        set(USE_${feature} ON CACHE BOOL "" FORCE)
    else()
        set(USE_${feature} OFF CACHE BOOL "" FORCE)
    endif()
endforeach()
foreach(feature MP3_MPG123 FLAC FFMPEG CMD MIDI_ADLMIDI MIDI_OPNMIDI MIDI_FLUIDSYNTH MIDI_NATIVE)
    set(USE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(AUDIO_CODECS_INSTALL_PATH "${OFE_AUDIOCODECS_ROOT}" CACHE PATH "" FORCE)
set(AUDIO_CODECS_REPO_PATH "${OFE_AUDIOCODECS_SOURCE}" CACHE PATH "" FORCE)
if(NOT OFE_AUDIOCODECS_ROOT)
    set(AUDIO_CODECS_REPO_PATH "" CACHE PATH "" FORCE)
endif()
set(MIXERX_ENABLE_LGPL ON CACHE BOOL "" FORCE)
set(MIXERX_ENABLE_GPL OFF CACHE BOOL "" FORCE)
set(DOWNLOAD_AUDIO_CODECS_DEPENDENCY OFF CACHE BOOL "" FORCE)
set(USE_MIDI ON CACHE BOOL "" FORCE)
set(USE_MIDI_TIMIDITY ON CACHE BOOL "" FORCE)
add_subdirectory("${OFE_MIXERX_SOURCE}" mixer)
include("${CMAKE_CURRENT_LIST_DIR}/MixerXFixes.cmake")
if(OFE_AUDIOCODECS_ROOT)
    target_compile_definitions(SDL2_mixer_ext PRIVATE "MODPLUG_HEADER=\"${OFE_AUDIOCODECS_ROOT}/include/libmodplug/modplug.h\"")
endif()
# SDL2's Android install exports its target twice. We ship binaries directly, so
# link its build output without exporting an SDL application/SDL2main dependency.
get_target_property(mixer_links SDL2_mixer_ext LINK_LIBRARIES)
set(build_links)
foreach(dependency IN LISTS mixer_links)
    if(dependency STREQUAL "SDL2::SDL2main")
        continue()
    elseif(dependency STREQUAL "SDL2::SDL2" OR dependency STREQUAL "SDL2")
        list(APPEND build_links "$<TARGET_FILE:SDL2>")
    else()
        list(APPEND build_links "${dependency}")
    endif()
endforeach()
set_property(TARGET SDL2_mixer_ext PROPERTY LINK_LIBRARIES "${build_links}")
add_dependencies(SDL2_mixer_ext SDL2)
if(NOT ANDROID AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(SDL2_mixer_ext PRIVATE alloca=__builtin_alloca)
endif()
target_include_directories(SDL2_mixer_ext PRIVATE "${OFE_SDL_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/sdl/include-config-$<CONFIG>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SOUNDSTRETCH OFF CACHE BOOL "" FORCE)
set(SOUNDTOUCH_DLL OFF CACHE BOOL "" FORCE)
set(OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${OFE_SOUNDTOUCH_SOURCE}" soundtouch)
if(ANDROID AND ANDROID_ABI STREQUAL "x86_64")
    # The API-23 software emulator faults on Clang's packed float-to-double load
    # when the small coefficient allocation ends at a page boundary. Keep the
    # explicit SSE audio kernels; only scalarize coefficient setup in this file.
    set_source_files_properties("${OFE_SOUNDTOUCH_SOURCE}/source/SoundTouch/FIRFilter.cpp"
        TARGET_DIRECTORY SoundTouch PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
endif()
add_library(ofe_audio SHARED AudioRuntime.cpp PCMStream.cpp)
target_compile_features(ofe_audio PUBLIC cxx_std_17)
target_include_directories(ofe_audio PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(ofe_audio PUBLIC SDL2 SDL2_mixer_ext PRIVATE SoundTouch)
add_library(ofe_sdl_mixer SHARED Mixer.cpp JNI.cpp)
target_compile_features(ofe_sdl_mixer PRIVATE cxx_std_17)
target_compile_options(ofe_sdl_mixer PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ofe_sdl_mixer PRIVATE ofe_audio SoundTouch)
foreach(target SDL2 SDL2_mixer_ext ofe_audio ofe_sdl_mixer)
    set_target_properties(${target} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(ANDROID)
    foreach(target SDL2 SDL2_mixer_ext ofe_audio ofe_sdl_mixer)
        target_link_options(${target} PRIVATE "-Wl,-z,max-page-size=16384")
    endforeach()
else()
    find_path(OFE_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(OFE_JNI_PLATFORM jni_md.h HINTS "$ENV{JAVA_HOME}/include/linux" "$ENV{JAVA_HOME}/include/darwin" REQUIRED)
    target_include_directories(ofe_sdl_mixer PRIVATE "${OFE_JNI_INCLUDE}" "${OFE_JNI_PLATFORM}")
    if(OFE_MIXER_TESTS)
        enable_testing()
        add_executable(ofe_mixer_test "${CMAKE_CURRENT_LIST_DIR}/../../../Android_Gradle/tests/sdl_mixer/MixerTest.cpp" Mixer.cpp)
        target_compile_features(ofe_mixer_test PRIVATE cxx_std_17)
        target_include_directories(ofe_mixer_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
        target_link_libraries(ofe_mixer_test PRIVATE ofe_audio SoundTouch)
        add_executable(ofe_pcm_test "${CMAKE_CURRENT_LIST_DIR}/../../../Android_Gradle/tests/sdl_mixer/PCMTest.cpp" AudioRuntime.cpp PCMStream.cpp)
        target_compile_features(ofe_pcm_test PRIVATE cxx_std_17)
        target_include_directories(ofe_pcm_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
        target_link_libraries(ofe_pcm_test PRIVATE SDL2 SDL2_mixer_ext SoundTouch)
        if(OFE_MIXER_SANITIZE)
            target_compile_options(ofe_pcm_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(ofe_pcm_test PRIVATE -fsanitize=address,undefined)
            target_compile_options(ofe_mixer_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(ofe_mixer_test PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()
