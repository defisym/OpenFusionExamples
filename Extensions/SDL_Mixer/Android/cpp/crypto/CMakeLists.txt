cmake_minimum_required(VERSION 3.22)
project(OFEMediaCrypto LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ofe_extensions "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT EXISTS "${ofe_extensions}/_DeLib/Encryption/CMakeLists.txt")
    get_filename_component(ofe_extensions "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
endif()
if(ANDROID)
    include("${CMAKE_CURRENT_LIST_DIR}/EnsureSDK.cmake")
endif()
function(ofe_add_encryption)
    # The Windows implementation and this JNI target compile the same sources.
    # Keep OpenSSL inside the owned JNI library rather than adding a global loader dependency.
    set(BUILD_SHARED_LIBS OFF)
    add_subdirectory("${ofe_extensions}/_DeLib/Encryption" "${CMAKE_CURRENT_BINARY_DIR}/encryption")
endfunction()
if(NOT TARGET encryption)
    ofe_add_encryption()
endif()
add_library(ofe_crypto SHARED JNI.cpp)
target_compile_features(ofe_crypto PRIVATE cxx_std_17)
target_link_libraries(ofe_crypto PRIVATE encryption)
set_target_properties(ofe_crypto PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(ANDROID)
    target_link_options(ofe_crypto PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
    find_path(OFE_CRYPTO_JNI jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(OFE_CRYPTO_JNI_PLATFORM jni_md.h HINTS "$ENV{JAVA_HOME}/include/linux" "$ENV{JAVA_HOME}/include/darwin" "$ENV{JAVA_HOME}/include/win32" REQUIRED)
    target_include_directories(ofe_crypto PRIVATE "${OFE_CRYPTO_JNI}" "${OFE_CRYPTO_JNI_PLATFORM}")
endif()
