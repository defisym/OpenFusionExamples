cmake_minimum_required(VERSION 3.22)
project(OFEMobileEncryption LANGUAGES CXX)
# One runtime dependency for every plugin. Consumers include only CAPI.h.
if(ANDROID AND OFE_OPENSSL_ROOT)
    set(OPENSSL_INCLUDE_DIR "${OFE_OPENSSL_ROOT}/include" CACHE PATH "Target crypto headers" FORCE)
    set(OPENSSL_CRYPTO_LIBRARY "${OFE_OPENSSL_ROOT}/lib/libcrypto.a" CACHE FILEPATH "Target static crypto inside common runtime" FORCE)
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(ofe_encryption SHARED ../CAPI.cpp)
target_compile_features(ofe_encryption PRIVATE cxx_std_17)
target_include_directories(ofe_encryption PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
target_compile_definitions(ofe_encryption PRIVATE ENCRYPTION_BUILD ENCRYPTION_SHARED _FILE_OFFSET_BITS=64)
target_link_libraries(ofe_encryption PRIVATE OpenSSL::Crypto)
set_target_properties(ofe_encryption PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(ANDROID)
    target_link_libraries(ofe_encryption PRIVATE dl)
    target_link_options(ofe_encryption PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
endif()
