cmake_minimum_required(VERSION 3.22)
project(ofe_ffmpeg LANGUAGES CXX)
set(OFE_FFMPEG_ROOT "" CACHE PATH "FFmpeg SDK containing include and lib/<Android ABI> (or lib on host)")
if(ANDROID)
    include("${CMAKE_CURRENT_LIST_DIR}/EnsureSDK.cmake")
endif()
if(ANDROID AND EXISTS "${OFE_FFMPEG_ROOT}/${ANDROID_ABI}/include/libavcodec/avcodec.h")
    set(OFE_FFMPEG_ROOT "${OFE_FFMPEG_ROOT}/${ANDROID_ABI}")
endif()
if(NOT EXISTS "${OFE_FFMPEG_ROOT}/include/libavcodec/avcodec.h")
    message(FATAL_ERROR "Set OFE_FFMPEG_ROOT to an FFmpeg SDK with Android MediaCodec enabled")
endif()
get_filename_component(OFE_EXTENSIONS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT EXISTS "${OFE_EXTENSIONS_ROOT}/_3rdLib/FFMpeg.h")
    get_filename_component(OFE_EXTENSIONS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(OFE_SHARED_ROOT "${OFE_EXTENSIONS_ROOT}/_3rdLib" CACHE PATH "Original shared C++ decoder sources")
set(OFE_DELIB_ROOT "${OFE_EXTENSIONS_ROOT}/_DeLib" CACHE PATH "Shared C++ utilities")
add_library(ofe_ffmpeg_core STATIC
    "${OFE_SHARED_ROOT}/FFMpeg.cpp"
    "${OFE_SHARED_ROOT}/FFMpegAdapter.cpp"
    "${OFE_SHARED_ROOT}/FFMpegAdapterAndroid.cpp")
target_include_directories(ofe_ffmpeg_core PUBLIC "${OFE_SHARED_ROOT}" "${OFE_DELIB_ROOT}")
set_target_properties(ofe_ffmpeg_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ofe_ffmpeg_core PUBLIC cxx_std_20)
target_compile_options(ofe_ffmpeg_core PRIVATE -Wall -Wextra -Wno-unknown-pragmas -Wno-unused-parameter)
target_include_directories(ofe_ffmpeg_core PUBLIC "${OFE_FFMPEG_ROOT}/include")
foreach(component avformat avcodec avfilter swscale swresample avutil)
    unset(OFE_${component} CACHE)
    find_library(OFE_${component} NAMES ${component}
        PATHS "${OFE_FFMPEG_ROOT}/lib/${ANDROID_ABI}" "${OFE_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    add_library(ofe_${component} SHARED IMPORTED)
    set_target_properties(ofe_${component} PROPERTIES IMPORTED_LOCATION "${OFE_${component}}")
    target_link_libraries(ofe_ffmpeg_core PUBLIC ofe_${component})
endforeach()
add_library(ofe_ffmpeg SHARED JNI.cpp
    "${OFE_EXTENSIONS_ROOT}/FFMpeg/CopyAdapter/CopyAdapter.cpp"
    "${OFE_EXTENSIONS_ROOT}/FFMpeg/CopyAdapter/CopyAdapterAndroid.cpp")
target_link_libraries(ofe_ffmpeg PRIVATE ofe_ffmpeg_core)
target_include_directories(ofe_ffmpeg PRIVATE "${OFE_EXTENSIONS_ROOT}/FFMpeg/CopyAdapter")
if(TARGET ofe_audio)
    target_link_libraries(ofe_ffmpeg PRIVATE ofe_audio)
    target_link_libraries(ofe_ffmpeg_core PUBLIC SDL2)
else()
    set(OFE_AUDIO_ROOT "" CACHE PATH "Shared Mixer X builds, containing <ABI>/lib/libofe_audio.so")
    get_filename_component(ofe_audio_include "${CMAKE_CURRENT_LIST_DIR}/../../../SDL_Mixer/Android/cpp" ABSOLUTE)
    if(NOT EXISTS "${ofe_audio_include}/AudioRuntime.h")
        set(ofe_audio_include "${CMAKE_CURRENT_LIST_DIR}/../../SDL_Mixer/cpp")
    endif()
    set(OFE_AUDIO_INCLUDE "${ofe_audio_include}" CACHE PATH "Shared AudioRuntime.h directory")
    unset(OFE_AUDIO_LIBRARY CACHE)
    find_library(OFE_AUDIO_LIBRARY NAMES ofe_audio PATHS
        "${OFE_AUDIO_ROOT}/${ANDROID_ABI}/lib" "${OFE_AUDIO_ROOT}/lib"
        NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    add_library(ofe_audio_shared SHARED IMPORTED)
    set_target_properties(ofe_audio_shared PROPERTIES IMPORTED_LOCATION "${OFE_AUDIO_LIBRARY}")
    target_link_libraries(ofe_ffmpeg PRIVATE ofe_audio_shared)
    target_include_directories(ofe_ffmpeg PRIVATE "${OFE_AUDIO_INCLUDE}")
    get_filename_component(ofe_audio_libdir "${OFE_AUDIO_LIBRARY}" DIRECTORY)
    get_filename_component(ofe_audio_abi "${ofe_audio_libdir}" DIRECTORY)
    set(OFE_SDL_INCLUDE "${ofe_audio_abi}/codecs-sdk/include/SDL2" CACHE PATH "Shared SDL headers")
    target_include_directories(ofe_ffmpeg_core PUBLIC "${OFE_SDL_INCLUDE}")
    find_library(OFE_SDL_LIBRARY NAMES SDL2 SDL2-2.0 PATHS "${ofe_audio_abi}/lib"
        NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    target_link_libraries(ofe_ffmpeg_core PUBLIC "${OFE_SDL_LIBRARY}")
endif()
if(ANDROID)
    target_link_options(ofe_ffmpeg PRIVATE "-Wl,-z,max-page-size=16384")
else()
    find_path(OFE_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(OFE_JNI_PLATFORM jni_md.h HINTS "$ENV{JAVA_HOME}/include/linux"
        "$ENV{JAVA_HOME}/include/darwin" "$ENV{JAVA_HOME}/include/win32" REQUIRED)
    target_include_directories(ofe_ffmpeg PRIVATE "${OFE_JNI_INCLUDE}" "${OFE_JNI_PLATFORM}")
endif()
if(ANDROID)
    target_sources(ofe_ffmpeg PRIVATE CopyAdapter.cpp)
    target_link_libraries(ofe_ffmpeg PRIVATE GLESv2 log)
endif()
